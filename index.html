// ==========================================
// Google Apps Script : Code.gs

function doGet(e) {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('FLIX CURATE - 영화 & 드라마 큐레이션')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1.0');
}

/**
 * Server-side TMDB proxy call if API Key is stored in Script Properties
 */
function fetchTmdbData(endpoint, params) {
  var apiKey = PropertiesService.getScriptProperties().getProperty('TMDB_API_KEY');
  if (!apiKey) {
    return { success: false, error: 'API Key not set in Script Properties.' };
  }
  
  var url = 'https://api.themoviedb.org/3' + endpoint + '?api_key=' + apiKey + '&language=ko-KR';
  if (params) {
    for (var key in params) {
      url += '&' + key + '=' + encodeURIComponent(params[key]);
    }
  }
  
  try {
    var response = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    return { success: true, data: JSON.parse(response.getContentText()) };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}
